Set-Location C:\path\to\CueBuilder
$listener = New-Object System.Net.HttpListener
$listener.Prefixes.Add('http://127.0.0.1:8765/')
$listener.Start()
Start-Process 'http://127.0.0.1:8765/'
while ($listener.IsListening) {
  $ctx = $listener.GetContext()
  $rel = [Uri]::UnescapeDataString($ctx.Request.Url.AbsolutePath)
  if ($rel -eq '/') { $rel = '/index.html' }
  $file = Join-Path (Get-Location) ($rel.TrimStart('/').Replace('/','\'))
  if (Test-Path $file) {
    $bytes = [IO.File]::ReadAllBytes($file)
    $ext = [IO.Path]::GetExtension($file)
    $ctx.Response.ContentType = switch ($ext) {
      '.html' {'text/html'} '.js' {'text/javascript'} '.css' {'text/css'} '.png' {'image/png'} default {'application/octet-stream'}
    }
    $ctx.Response.OutputStream.Write($bytes, 0, $bytes.Length)
  } else { $ctx.Response.StatusCode = 404 }
  $ctx.Response.Close()
}
